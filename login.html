function login(){ const email=document.getElementById('email').value.trim().toLowerCase(); const pass=document.getElementById('password').value; let users=JSON.parse(localStorage.getItem('allUsers'))||[]; let u=users.find(x=>x.email.toLowerCase()===email && x.password===pass); if(u){ localStorage.setItem('currentUser', JSON.stringify(u)); // FORCE CORRECT REDIRECT if(u.role==='vendor'){ window.location.href='vendor.html'; } else if(u.role==='delivery'){ window.location.href='delivery.html'; } else { window.location.href='index.html'; } } else { alert('Invalid email or password\n\nIf you are on Vercel first time, please Sign Up first.'); } }
